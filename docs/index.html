Your first Glyph program (RP2040 + SSD1306)

with RP.GPIO;
with RP.I2C_Master;
with RP.Device;
with Glyph.Display.SSD1306;

procedure Main is
   I2C  : RP.I2C_Master.I2C_Master_Port renames RP.Device.I2CM_0;
   SDA  : RP.GPIO.GPIO_Point renames RP.Device.GP8;
   SCL  : RP.GPIO.GPIO_Point renames RP.Device.GP9;

   OLED : Glyph.Display.SSD1306.Display_T := Glyph.Display.SSD1306.Create;
begin
   -- Configure RP2040 I2C pins with pull-ups
   SDA.Configure (RP.GPIO.Output, RP.GPIO.Pull_Up, RP.GPIO.I2C, Schmitt => True);
   SCL.Configure (RP.GPIO.Output, RP.GPIO.Pull_Up, RP.GPIO.I2C, Schmitt => True);
   I2C.Configure (Baudrate => 400_000);

   -- Connect display to hardware bus
   OLED.Init (I2C'Access);
   OLED.Clear;

   -- Draw directly to display instance
   OLED.Paint_Rectangle (0, 0, 128, 64);
   OLED.Paint_Circle (Center_X => 64, Center_Y => 32, Radius => 18);
   OLED.Paint_Text (X => 48, Y => 29, Text => "GLYPH");

   -- Flush frame to physical OLED panel
   OLED.Render;
end Main;

Every drawing call operates directly on the display instance. All internal buffers are statically sized with zero heap allocation.
